\begin{MyRecipe}{Walnussbrot}{\Calc{1}{\x} Brot}{\SI{12}{\hour} + \SI{15}{\minuteprime} + \SI{2,5}{\hour} + \SI{45}{\minuteprime}}

	\ingredient[\Calc{325}{\x}]{\si{\gram}} {Buttermilch}
	\ingredient[\Calc{125}{\x}]{\si{\gram}} {Roggenvollkornmehl}
	\Step{Vorteig}
	Buttermilch mit Roggenvollkornmehl mischen und  über Nachst (ca \SI{12}{\hour}) abgedeckt bei Zimmertemperatur ruhen lassen.\par\bigskip

	\ingredient[\Calc{375}{\x}]{\si{\gram}} {Ruchmehl}
	\ingredient[\Calc{115}{\x}]{\si{\gram}} {Wasser (\SI{50}{\celsius})}
	\ingredient[\Calc{100}{\x}]{\si{\gram}} {Kartoffel}
	\ingredient[\Calc{10}{\x}]{\si{\gram}} {Salz}
	\ingredient[\Calc{100}{\x}]{\si{\gram}} {Walnüsse, geröstet}
	\ingredient[\Calc{5}{\x}]{\si{\gram}} {Hefe}

	\Step{Hauptteig}
	Vorteig, Ruchmehl, Wasser, geriebene Katoffel, Salz und Hefe \SI{5}{\minuteprime} auf Stufe 1 und anschließend \SI{7}{\minuteprime} auf Stufe 2 in Maschine kneten lassen. Walnüsse währenddessen anrösten und in fertig gekneteten Teig untermischen.\par\bigskip
	
	\Step{Stockgare}
	Hauptteig \SI{2}{\hour} auf das doppelte im Backofen mit Dampf gehen lassen. Nach etwa \SI{30}{\minuteprime} und \SI{60}{\minuteprime} dehnen und falten.\par\bigskip
	
	\Step{Stückgare}
	Hauptteig langwirken und mit Schluss nach oben \SI{45}{\minuteprime} im Gärkröbchen gehen lassen.\par\bigskip

	\Step{Backen}
	Teig auf ein Brett mit etwas Mehl stürzen, längsseitig tief einschneiden und mit Schluss nach unten \SI{50}{\minuteprime} bei \SI{250}{\celsius}, fallend auf \SI{220}{\celsius} mit Dampf backen.\par\bigskip

\end{MyRecipe}
