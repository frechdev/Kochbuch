\begin{MyRecipe}{Apfel-Quarkbällchen}{\Calc{20}{\x} - \Calc{25}{\x} Stück}{\SI{15}{\minuteprime}+\SI{20}{\minuteprime}}
	
	\ingredient[\Calc{2}{\x}]{} {Petersilienwurzeln}


	\Step{Backofen vorbereiten}
	Backofen auf \SI{180}{\degreeCelsius} (Ober-/Unterhitze) vorheizen. Backblech mit Backpapier auslegen.
	
	\ingredient[\Calc{0.2}{\x}]{\si{\kilo\gram}}{Magerquark}
	\ingredient[\Calc{0.3}{\x}]{\si{\kilo\gram}}{Mehl}
	\ingredient[\Calc{0.1}{\x}]{\si{\liter}}{Sonnenblumenöl}
	\ingredient[\Calc{0.11}{\x}]{\si{\kilo\gram}}{Zucker}
	\ingredient[\Calc{1}{\x}]{\si{\Paeckchen}}{Vanillezucker}
	\ingredient[\Calc{0.5}{\x}]{\si{\Paeckchen}}{Backpulver}
	\ingredient[\Calc{2.5}{\x}]{}{Äpfel}

	\Step{Teig}
	Quark, Öl, Zucker und Vanillezucker in Schüssel geben und verrühren.\\
	Mehl mit Backpulver darüber sieben und alles mit einem Holzlöffel zu einem weichen Teig verarbeiten.
	Apfel schälen, vierteln, entkernen, in grobe Stücke schneiden und in den Teig einarbeiten.\\
	Der Teig sollte schön klebrig sein.\par\bigskip

	\Step{Bällchen formen}
	Aus der Teigmischung entsprechend der Menge etwa 3 Finger breite, kleine Bällchen formen und auf das vorbereitete Backblech mit Backpapier setzen. \par\bigskip
	
	\Step{Backen}
	Im vorgeheizten Ofen etwa  \SI{20}{\minuteprime} bis  \SI{25}{\minuteprime} backen, bis die Quarkbällchen goldbraun sind.\par\bigskip
	
	\ingredient[\Calc{40}{\x}]{\si{\gram}}{Zucker}
	\ingredient[\Calc{1}{\x}]{\si{\Teeloeffel}}{Zimt}

	\Step{Überzug}
	Für den Überzug den Zucker mit dem Zimt in einer Schale vermischen. Die Apfel-Quarkbällchen aus dem Ofen nehmen und noch warm in der Zimtzucker-Mischung wälzen. Auf ein
	Kuchengitter setzen und auskühlen lassen.

	
	
	
\end{MyRecipe}
