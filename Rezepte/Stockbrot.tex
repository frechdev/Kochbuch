\begin{MyRecipe}{Stockbrot}{\Calc{10}{\x} Stück}{\SI{20}{\minuteprime} + \SI{60}{\minuteprime}}
	\ingredient[\Calc{500}{\x}]{\si{\gram}} {Mehl (405)}
	\ingredient[\Calc{0.25}{\x}]{\si{\liter}} {Wasser, lauwarm}
	\ingredient[0.5]{} {Frischhefe}
	\ingredient[\Calc{2}{\x}]{\si{\Teeloeffel}} {Zucker}
	\ingredient[\Calc{2}{\x}]{\si{\Teeloeffel}} {Salz}
	\ingredient[\Calc{1}{\x}]{\si{\Essloeffel}} {Olivenöl}
	
	\Step{Ansetzen}
	Hefe zusammen mit dem Zucker im lauwarmen Wasser verrühren, sodass die Hefe aktiviert wird. Anschließend zusammen mit Mehl, Olivenöl, Salz sowie der lauwarmen Milch gut ca. \SI{10}{\minuteprime} mit Küchenmaschine verkneten.\par\bigskip
	
	\Step{Gehen lassen}
	Teig mit etwas Mehl bestäumen und abgedeckt ihn für ca. \SI{30}{\minuteprime} bis \SI{60}{\minuteprime} einen warmen Ort gehen lassen.
	
	
\end{MyRecipe}
